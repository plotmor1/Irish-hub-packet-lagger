local function clampScale(s)
	local stepped = math.floor(((s - 0.5) / 0.1) + 0.5) * 0.1 + 0.5
	return math.clamp(stepped, 0.5, 1.5)
end

local function getIntensity()
	if mode == "PC" then return 50000 end
	if mode == "Mobile" then return 25000 end
	return customIntensity
end

local function saveSettings()
	pcall(function()
		writefile(SETTINGS_FILE, HttpService:JSONEncode({
			Tab = mode,
			GuiSize = guiScale,
			Toggle = lagOn,
			Keybind = keybind and { Name = keybind.Name, Type = keybindType } or nil,
			SpeedEnabled = speedOn,
			SpeedStuds = speedStuds,
			SpeedInterval = speedInterval,
			CustomIntensity = customIntensity,
		}))
	end)
end

local function loadSettings()
	if not isfile or not isfile(SETTINGS_FILE) then return end
	local ok, raw = pcall(readfile, SETTINGS_FILE)
	if not ok or not raw or raw == "" then return end
	local ok2, data = pcall(function() return HttpService:JSONDecode(raw) end)
	if not ok2 or type(data) ~= "table" then return end

	if data.Tab == "PC" or data.Tab == "Mobile" or data.Tab == "Custom" then
		mode = data.Tab
	end
	if data.GuiSize then
		guiScale = clampScale(tonumber(data.GuiSize) or 0.7)
	end
	if data.Toggle ~= nil then
		lagOn = data.Toggle
	end
	if data.Keybind then
		local name, typ = data.Keybind, "Keyboard"
		if type(data.Keybind) == "table" then
			name = data.Keybind.Name
			typ = data.Keybind.Type or "Keyboard"
		end
		local kc = name and Enum.KeyCode[name]
		if kc then
			keybind = kc
			keybindType = typ
		end
	end
	if data.SpeedEnabled ~= nil then speedOn = data.SpeedEnabled end
	if data.SpeedStuds then speedStuds = tonumber(data.SpeedStuds) or 15 end
	if data.SpeedInterval then speedInterval = tonumber(data.SpeedInterval) or 0.06 end
	if data.CustomIntensity then
		customIntensity = math.clamp(tonumber(data.CustomIntensity) or 75000, 1, 500000)
	end
end

loadSettings()

-- Packet lag core (Packages.Net child index 230)
---------------------------------------------------------------------
local lagRemote = nil

local function resolveRemote()
	local ok, net = pcall(function()
		return RS:WaitForChild("Packages"):WaitForChild("Net")
	end)
	if ok and net then
		lagRemote = net:GetChildren()[230]
	end
	return lagRemote
end

resolveRemote()

local function fireLagBatch(count)
	if not lagRemote then
		resolveRemote()
		if not lagRemote then return end
	end
	local payload = table.create(count, function() end)
	for _ = 1, 3 do
		pcall(function()
			lagRemote:FireServer(payload)
		end)
	end
end

local function lagLoop()
	while lagRunning do
		local n = getIntensity()
		task.spawn(function()
			fireLagBatch(n)
		end)
		task.wait(0.01)
	end
end

local function startLag()
	if lagThread and coroutine.status(lagThread) ~= "dead" then
		return
	end
	lagRunning = true
	lagStartClock = os.clock()
	lagThread = coroutine.create(lagLoop)
	coroutine.resume(lagThread)
end

local function stopLag()
	lagRunning = false
	if lagThread then
		pcall(function() coroutine.close(lagThread) end)
		lagThread = nil
	end
end

track(UIS.InputBegan:Connect(function(input, gp)
	if capturingKey then
		if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode ~= Enum.KeyCode.Unknown then
			keybind = input.KeyCode
			keybindType = "Keyboard"
			keybindBtn.Text = input.KeyCode.Name
			keybindStroke.Color = GREEN
			capturingKey = false
			saveSettings()
			return
		end
		if input.UserInputType == Enum.UserInputType.Gamepad1 and input.KeyCode ~= Enum.KeyCode.Unknown then
			keybind = input.KeyCode
			keybindType = "Gamepad"
			keybindBtn.Text = input.KeyCode.Name
			keybindStroke.Color = GREEN
			capturingKey = false
			saveSettings()
			return
		end
		return
	end

	if gp or not keybind then return end
	if input.KeyCode ~= keybind then return end
	if keybindType == "Keyboard" and input.UserInputType ~= Enum.UserInputType.Keyboard then return end
	if keybindType == "Gamepad" and input.UserInputType ~= Enum.UserInputType.Gamepad1 then return end

	setLag(not lagOn)
end))

local speedAcc = 0
track(RunService.Heartbeat:Connect(function(dt)
	if not lagRunning or not speedOn then return end
	speedAcc += dt
	if speedAcc < speedInterval then return end
	speedAcc = 0

	local char = LP.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hum or not hrp then return end
	local dir = hum.MoveDirection
	if dir.Magnitude <= 0 then return end
	hrp.CFrame = hrp.CFrame + dir * speedStuds
end))

local fpsFrames, fpsTime = 0, 0
track(RunService.Heartbeat:Connect(function(dt)
	if lagRunning then
		local t = math.floor(os.clock() - lagStartClock)
		timerLbl.Text = string.format("%02d:%02d", math.floor(t / 60), t % 60)
	else
		timerLbl.Text = "00:00"
	end
	fpsFrames += 1
	fpsTime += dt
	if fpsTime >= 0.5 then
		fpsLbl.Text = "FPS: " .. tostring(math.floor(fpsFrames / fpsTime + 0.5))
		fpsFrames, fpsTime = 0, 0
	end
end))

local dragging, dragStart, frameStart
track(header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		frameStart = main.Position
	end
end))
track(UIS.InputChanged:Connect(function(input)
	if not dragging then return end
	if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
	local d = input.Position - dragStart
	main.Position = UDim2.new(frameStart.X.Scale, frameStart.X.Offset + d.X, frameStart.Y.Scale, frameStart.Y.Offset + d.Y)
end))
track(UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end))
